cmake_minimum_required(VERSION 3.28)
option(VEIL_BUILD_OBS "Build native OBS filter (requires OBS SDK)" OFF)
if(VEIL_BUILD_OBS)
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
endif()
project(veil-guard VERSION 0.1.0 LANGUAGES C CXX)
if(VEIL_BUILD_OBS)
  include(compilerconfig)
  include(defaults)
  include(helpers)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
include(CTest)
add_library(veil-core STATIC src/guard-core.c)
set_target_properties(veil-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(veil-core PUBLIC src)
add_executable(veil-core-test tests/test-core.c)
target_link_libraries(veil-core-test PRIVATE veil-core)
add_test(NAME sampling-timeline COMMAND veil-core-test)
if(VEIL_BUILD_OBS)
  find_package(libobs REQUIRED)
  add_library(veil-guard MODULE src/veil-filter.cpp src/detector-client.cpp)
  target_link_libraries(veil-guard PRIVATE veil-core OBS::libobs)
  if(WIN32)
    target_link_libraries(veil-guard PRIVATE winhttp)
  else()
    find_package(CURL REQUIRED)
    target_link_libraries(veil-guard PRIVATE CURL::libcurl)
  endif()
  set_target_properties(veil-guard PROPERTIES PREFIX "")
  set_target_properties_plugin(veil-guard PROPERTIES OUTPUT_NAME "veil-guard")
  install(TARGETS veil-guard LIBRARY DESTINATION obs-plugins/64bit RUNTIME DESTINATION obs-plugins/64bit)
  install(DIRECTORY data/ DESTINATION data/obs-plugins/veil-guard)
endif()
